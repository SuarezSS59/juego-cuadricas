        let selectedCards = [];
        let score = 0;
        let timerSeconds = 0;
        let timerInterval = null;
        let currentRoom = 1;

        // Generar botones para los 10 grupos
        const roomsContainer = document.getElementById('rooms-container');
        for (let i = 1; i <= 10; i++) {
            const btn = document.createElement('button');
            btn.className = 'btn-room';
            btn.innerText = `Grupo ${i}`;
            btn.onclick = () => enterRoom(i);
            roomsContainer.appendChild(btn);
        }

        // Generador pseudo-aleatorio basado en el número de grupo
        function seededRandom(seed) {
            let x = Math.sin(seed++) * 10000;
            return x - Math.floor(x);
        }

        function shuffleSeeded(array, seed) {
            let m = array.length, t, i;
            while (m) {
                i = Math.floor(seededRandom(seed++) * m--);
                t = array[m];
                array[m] = array[i];
                array[i] = t;
            }
            return array;
        }

        function enterRoom(roomNumber) {
            currentRoom = roomNumber;
            document.getElementById('display-room').innerText = roomNumber;
            document.getElementById('room-select').style.display = 'none';
            document.getElementById('game-screen').style.display = 'block';

            initGame();
        }

        function leaveRoom() {
            clearInterval(timerInterval);
            document.getElementById('game-screen').style.display = 'none';
            document.getElementById('room-select').style.display = 'block';
        }

        function initGame() {
            const board = document.getElementById('board');
            board.innerHTML = '';
            selectedCards = [];
            score = 0;
            timerSeconds = 0;
            document.getElementById('score').innerText = score;
            document.getElementById('timer').innerText = '0s';

            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                timerSeconds++;
                document.getElementById('timer').innerText = `${timerSeconds}s`;
            }, 1000);

            // Mezcla única fijada por el número del grupo
            const shuffled = shuffleSeeded([...cardsData], currentRoom * 100);

            shuffled.forEach(card => {
                const cardEl = document.createElement('div');
                cardEl.className = 'card';
                cardEl.dataset.family = card.family;

                cardEl.innerHTML = `
                    <span class="card-type type-${card.type}">${card.label}</span>
                    <div class="card-title">${card.title}</div>
                    <div class="card-content">${card.content}</div>
                `;

                cardEl.onclick = () => selectCard(cardEl);
                board.appendChild(cardEl);
            });
        }

        function selectCard(cardEl) {
            if (selectedCards.length < 2 && !cardEl.classList.contains('selected')) {
                cardEl.classList.add('selected');
                selectedCards.push(cardEl);

                if (selectedCards.length === 2) {
                    checkMatch();
                }
            }
        }

        function checkMatch() {
            const [card1, card2] = selectedCards;

            if (card1.dataset.family === card2.dataset.family) {
                setTimeout(() => {
                    card1.classList.remove('selected');
                    card2.classList.remove('selected');
                    card1.classList.add('matched');
                    card2.classList.add('matched');
                    score += 10;
                    document.getElementById('score').innerText = score;
                    selectedCards = [];

                    if (document.querySelectorAll('.card.matched').length === cardsData.length) {
                        clearInterval(timerInterval);
                        alert(`¡Felicidades Grupo ${currentRoom}! Completaron el juego en ${timerSeconds} segundos.`);
                    }
                }, 300);
            } else {
                setTimeout(() => {
                    card1.classList.remove('selected');
                    card2.classList.remove('selected');
                    selectedCards = [];
                }, 700);
            }
        }
